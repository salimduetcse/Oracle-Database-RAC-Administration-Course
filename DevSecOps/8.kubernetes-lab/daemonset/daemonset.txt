demo-daemonset.yaml
--------------------------------

apiVersion: apps/v1
kind: DaemonSet
metadata:
  name: hello-daemon
  namespace: default
spec:
  selector:
    matchLabels:
      app: hello-daemon
  template:
    metadata:
      labels:
        app: hello-daemon
    spec:
      containers:
      - name: hello
        image: busybox
        command: ["sh", "-c"]
        args:
          - while true; do echo "Hello from $(hostname)" | nc -lk -p 8080; done
        ports:
        - containerPort: 8080
        resources:
          requests:
            cpu: "50m"
            memory: "64Mi"
          limits:
            cpu: "100m"
            memory: "128Mi"
